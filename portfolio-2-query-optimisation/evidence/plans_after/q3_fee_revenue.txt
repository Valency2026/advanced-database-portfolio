SELECT date_trunc('day', created_at)::date AS day,
       count(*)                            AS fee_count,
       sum(amount)                         AS fee_revenue_ngn
FROM transactions
WHERE txn_type = 'fee'
  AND created_at >= '2026-10-01' AND created_at < '2026-11-01'
GROUP BY 1
ORDER BY 1;
